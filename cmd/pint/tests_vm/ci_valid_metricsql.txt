# pint ci passes when a branch only adds valid MetricsQL rules.
mkdir testrepo
cd testrepo
exec git init --initial-branch=main .

cp ../src/v1.yml rules.yml
cp ../src/.pint.hcl .
env GIT_AUTHOR_NAME=pint
env GIT_AUTHOR_EMAIL=pint@example.com
env GIT_COMMITTER_NAME=pint
env GIT_COMMITTER_EMAIL=pint@example.com
exec git add .
exec git commit -am 'import rules and config'

exec git checkout -b v2
cp ../src/v2.yml rules.yml
exec git commit -am 'v2'

exec pint --no-color ci
! stdout .
stderr 'level=INFO msg="Checking Prometheus rules" entries=2 workers=10 online=true'
! stderr 'Problems found'
! stderr 'Bug:|Fatal:|Warning:'

-- src/v1.yml --
- record: job:pint_vm_e2e_requests:rate5m
  expr: 'sum(rate(pint_vm_e2e_requests_total{job="api"}[5m])) by (job)'

-- src/v2.yml --
- record: job:pint_vm_e2e_requests:rate5m
  expr: 'sum(rate(pint_vm_e2e_requests_total{job="api"}[5m])) by (job)'
- record: job:pint_vm_e2e_requests:rollup
  expr: 'WITH (f(m) = rollup_rate(m[5m])) sum(f(pint_vm_e2e_requests_total{job="api"})) by (job)'

-- src/.pint.hcl --
ci {
  baseBranch = "main"
}
parser {
  relaxed        = [".*"]
  query_language = "metricsql"
}
prometheus "vm" {
  uri     = ENV_PINT_VM_URL
  backend = "victoriametrics"
  uptime  = "pint_vm_e2e_up"
  timeout = "30s"
}
