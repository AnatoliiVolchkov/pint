# group/interval reads VictoriaMetrics flags that change how far back instant queries look,
# here -search.minStalenessInterval covers the 10m interval. Offline it uses the defaults.
http response vm ^/flags$ 200 -search.minStalenessInterval="30m0s"

exec pint --no-color --offline lint rules
