# --external-label accepts comma separated labels.
exec pint --no-color --external-label account=alpha,dc=test lint rules
! stdout .
stderr 'msg="Using external labels from the command line" labels=\{"account":"alpha","dc":"test"\} servers=1'
! stderr 'alerts/external_labels'
stderr 'Warning: conflicting labels \(labels/conflict\)'
stderr '`dc="test"` is an external label of `vm` VictoriaMetrics server'

# The flag can be repeated.
exec pint --no-color --external-label account=alpha --external-label dc=test lint rules
stderr 'labels=\{"account":"alpha","dc":"test"\} servers=1'
! stderr 'alerts/external_labels'

# Labels can come from the PINT_EXTERNAL_LABELS environment variable.
env PINT_EXTERNAL_LABELS=account=alpha
exec pint --no-color lint rules
stderr 'labels=\{"account":"alpha"\} servers=1'
! stderr 'alerts/external_labels'
! stderr 'labels/conflict'

# The flag takes precedence over the environment variable.
