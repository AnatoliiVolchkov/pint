# MetricsQL turns rate(sum(foo)) and sum(foo)[1h] into implicit subqueries,
# VictoriaMetrics rejects them with -search.disableImplicitConversion.
exec pint --no-color lint rules
! stdout .
cmp stderr stderr.txt

# PromQL doesn't allow them at all.
! exec pint --no-color -c promql.hcl lint rules
! stdout .
stderr 'level=INFO msg="Problems found" Fatal=2'
