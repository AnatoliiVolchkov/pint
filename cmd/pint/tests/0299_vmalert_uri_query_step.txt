# With vmalertURI pint reads -datasource.queryStep from vmalert, here 20m, which is shorter
# than the 30m interval of the writer group. Offline vmalert isn't queried and the default
# 5m query step is used.
http response vm ^/flags$ 200 -retentionPeriod="1y"
http start vm 127.0.0.1:7298
http response vmalert ^/flags$ 200 -datasource.queryStep="20m0s"
http response vmalert ^/api/v1/rules$ 200 {"status":"success","data":{"groups":[{"name":"writer","file":"rules/1.yml"},{"name":"readers","file":"rules/1.yml"}]}}
http start vmalert 127.0.0.1:7299

exec pint --no-color -e group/interval lint --json report.json rules
! stdout .
cmp stderr online.txt
grep 'the query step, vmalert -datasource.queryStep read from http://127.0.0.1:7299 \(20m\)' report.json

exec pint --no-color --offline lint rules
! stdout .
cmp stderr offline.txt
